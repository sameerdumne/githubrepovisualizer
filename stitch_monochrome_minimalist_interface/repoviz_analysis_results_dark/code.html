import React, { useEffect, useState } from 'react';
import { Engine, Scene } from 'repo-viz-core';

/**
 * @description Advanced Analysis Visualizer component.
 * Performs real-time complexity mapping and dependency resolution.
 */
export const Visualizer: React.FC<VisualizerProps> = ({ data, config }) => {
  const [metrics, setMetrics] = useState<AnalysisMetrics | null>(null);
  
  useEffect(() => {
    const runAnalysis = async () => {
      try {
        const engine = new Engine(config.token);
        const results = await engine.analyze(data.repositoryId);
        
        // Optimize data for visualization
        const optimized = results.nodes.map(node => ({
          id: node.id,
          weight: Math.log(node.size) * config.scaleFactor,
          color: node.impact > 0.8 ? '#FFFFFF' : '#404040'
        }));

        setMetrics(optimized);
      } catch (error) {
        console.error('Analysis failed', error);
      }
    };

    runAnalysis();
  }, [data.repositoryId, config]);

  return (
    <div className="viewport-container">
      {metrics && <Scene data={metrics} />}
    </div>
  );
};
